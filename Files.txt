    public partial class UserControl1 : UserControl
    {
        public event EventHandler OnDataChanged;

        public UserControl1()
        {
            InitializeComponent();
            AttachChangeEvents(this.Controls);
        }

        private void UserControl1_Load(object sender, EventArgs e)
        {
            SetPlaceholder(txtProductName, "Nhập tên sản phẩm...");
            SetPlaceholder(txtServingUnit, "VD: bịch, phần, gói, hộp,...");

            SetCodeEditor(codeEditor);
        }

        private void AttachChangeEvents(Control.ControlCollection controls)
        {
            foreach (Control ctrl in controls)
            {
                if (ctrl.Name == "codeEditor")
                    continue;

                if (ctrl is TextBox txt)
                {
                    //assign the Event Changed to public all event (for text input)
                    txt.TextChanged += AnyControl_ValueChanged;
                }
                else if (ctrl is NumericUpDown nud)
                {
                    //assign the event value changed to public all event (for numeric up down input)
                    nud.ValueChanged += AnyControl_ValueChanged;
                }

                if (ctrl.HasChildren)
                {
                    //recursively attach events to child controls
                    AttachChangeEvents(ctrl.Controls);
                }
            }



        }

        private void AnyControl_ValueChanged(object sender, EventArgs e)
        {
            NutritionModel model = GetInfo();
            if (string.IsNullOrWhiteSpace(model.ProductName) 
                && string.IsNullOrWhiteSpace(model.ServingUnit)
                && model.Calories <= 0
                )
            {
                return;
            }
            else
            {
                new Python().MakeScript("../../../_python/script/Script.py");
            }

            OnDataChanged?.Invoke(this, EventArgs.Empty);
        }





        public void SetCodeEditor(FastColoredTextBox codeEditor)
        {
            codeEditor.Language = Language.Custom;  // For Python
            codeEditor.AutoCompleteBrackets = true;
            codeEditor.CommentPrefix = "#";
            codeEditor.Range.tb.LeftBracket = '(';  // Bracket config doesn't work in PythonSyntax.xml
            codeEditor.Range.tb.RightBracket = ')';
            codeEditor.Range.tb.LeftBracket2 = '[';
            codeEditor.Range.tb.RightBracket2 = ']';

            try
            {
                string tempXmlPath = Path.Combine(Path.GetTempPath(), "PythonSyntax_temp.xml");
                string xmlContent = System.Text.Encoding.UTF8.GetString(Properties.Resources.PythonSyntax);
                File.WriteAllText(tempXmlPath, xmlContent, System.Text.Encoding.UTF8);
                codeEditor.DescriptionFile = tempXmlPath;
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Lỗi nạp cấu hình ngôn ngữ: {ex.Message}", "Lỗi Resource", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }

            codeEditor.OnTextChanged();
        }




        public void SetPlaceholder(TextBox textBox, string placeholderText)
        {
            textBox.Text = placeholderText;
            textBox.ForeColor = Color.Gray;
            textBox.Enter += (s, e) =>
            {
                if (textBox.Text == placeholderText)
                {
                    textBox.Text = "";
                    textBox.ForeColor = Color.Black;
                }
            };
            textBox.Leave += (s, e) =>
            {
                if (string.IsNullOrWhiteSpace(textBox.Text))
                {
                    textBox.Text = placeholderText;
                    textBox.ForeColor = Color.Gray;
                }
            };
        }


        public NutritionModel GetInfo()
        {
            return new NutritionModel
            {
                ProductName = txtProductName.Text == "Nhập tên sản phẩm..." ? "" : txtProductName.Text.Trim(),
                ServingUnit = txtServingUnit.Text == "VD: bịch, phần, gói, hộp,..." ? "" : txtServingUnit.Text.Trim(),
                Code = codeEditor.Text.Trim(),

                ServingCount = nudServingCount.Value,
                ServingWeight = nudServingWeight.Value,
                Calories = nudCalories.Value,

                TotalFat = nudTotalFat.Value,
                SaturatedFat = nudSaturatedFat.Value,
                TransFat = nudTransFat.Value,

                Cholesterol = nudCholesterol.Value,
                Sodium = nudSodium.Value,

                TotalCarbohydrate = nudTotalCarbs.Value,
                DietaryFiber = nudDietaryFiber.Value,
                TotalSugars = nudTotalSugars.Value,
                AddedSugars = nudAddedSugars.Value,
                Protein = nudProtein.Value,

                VitaminD = nudVitaminD.Value,
                Calcium = nudCalcium.Value,
                Iron = nudIron.Value,
                Potassium = nudPotassium.Value
            };
        }

"""Generate FDA-style Nutrition Facts label images from structured data."""

from typing import List, Optional

from PIL import Image, ImageDraw

from nutrition_facts.constants import (
    COLOR_WHITE,
    DEFAULT_DPI,
    DEFAULT_HEIGHT,
    DEFAULT_WIDTH,
    FDA_AMOUNT_PT,
    FDA_BOX_PT,
    FDA_CALORIES_LABEL_PT,
    FDA_CALORIES_VALUE_PT,
    FDA_DV_HEADER_PT,
    FDA_FOOTNOTE,
    FDA_FOOTNOTE_LEADING_PT,
    FDA_FOOTNOTE_PT,
    FDA_HAIRLINE_PAD_PT,
    FDA_INDENT_ADDED_PT,
    FDA_INDENT_SUB_PT,
    FDA_INSET_PT,
    FDA_MICRO_LEADING_PT,
    FDA_MICRO_PT,
    FDA_NUTRIENT_LEADING_PT,
    FDA_NUTRIENT_PT,
    FDA_RULE_HAIRLINE_PT,
    FDA_RULE_HEAVY_PT,
    FDA_RULE_MEDIUM_PT,
    FDA_SERVING_SIZE_PT,
    FDA_SERVINGS_LEADING_PT,
    FDA_SERVINGS_PT,
    FDA_TITLE,
    LABEL_WIDTH_INCHES,
    POINTS_PER_INCH,
    RGB_WHITE,
)
from nutrition_facts.models import Nutrient, NutritionData
from nutrition_facts.renderer import (
    draw_border,
    draw_horizontal_line,
    draw_nutrient_row,
    draw_text,
    draw_text_right,
    fit_font_size,
    measure_text,
    wrap_text,
)
from nutrition_facts.utils import (
    display_nutrient_name,
    ensure_parent_dir,
    format_daily_value,
    format_nutrient_value,
    format_servings,
    load_font,
    needs_full_hairline_before,
    nutrient_role,
    partition_nutrients,
)


class NutritionFactsGenerator:
    """Create a US FDA vertical Nutrition Facts label from NutritionData."""

    def __init__(
        self,
        width: int = DEFAULT_WIDTH,
        height: int = DEFAULT_HEIGHT,
        dpi: int = DEFAULT_DPI,
    ) -> None:
        if width < 200 or height < 200:
            raise ValueError("width and height must be at least 200 pixels")

        if dpi < 72:
            raise ValueError("dpi must be at least 72")

        self.width = width
        self.height = height
        self.dpi = dpi

        self._px_per_pt = width / (
            LABEL_WIDTH_INCHES * POINTS_PER_INCH
        )

        self._inset = self._pt(FDA_INSET_PT)
        self._box = max(1, self._pt(FDA_BOX_PT))

        self._left = self._box + self._inset
        self._right = width - self._box - self._inset
        self._content_width = self._right - self._left

        self._cursor_y = 0
        self._title_px = 0

    def _pt(self, points: float) -> int:
        if points <= 0:
            return 0

        px = points * self._px_per_pt
        return max(1, round(px))

    def generate(
        self,
        data: NutritionData,
        output_path: str,
    ) -> str:
        """Validate data, render a label image, save it, and return the path."""

        self._validate_data(data)

        img = self._create_blank_image()

        self._draw_header(img, data)
        self._draw_calories(img, data.calories)
        self._draw_nutrient_table(img, data.nutrients)
        self._draw_footer(img, data)

        img = self._crop_and_frame(img)

        return self._save_image(img, output_path)

    def _validate_data(self, data: NutritionData) -> None:
        if not isinstance(data, NutritionData):
            raise TypeError(
                "data must be a NutritionData instance"
            )

        payload = (
            data.model_dump()
            if hasattr(data, "model_dump")
            else data.dict()
        )

        if hasattr(NutritionData, "model_validate"):
            NutritionData.model_validate(payload)
        else:
            NutritionData.parse_obj(payload)

    def _create_blank_image(self) -> Image.Image:
        canvas_h = max(self.height, 2400)

        img = Image.new(
            "RGB",
            (self.width, canvas_h),
            RGB_WHITE,
        )

        self._cursor_y = self._box + self._inset

        return img

    def _rule(
        self,
        img: Image.Image,
        thickness_pt: float,
        x: Optional[int] = None,
        width: Optional[int] = None,
    ) -> None:
        x0 = self._left if x is None else x
        w = self._content_width if width is None else width

        thickness = self._pt(thickness_pt)

        draw_horizontal_line(
            img,
            self._cursor_y,
            w,
            thickness=thickness,
            x=x0,
        )

        self._cursor_y += thickness

    def _draw_header(
        self,
        img: Image.Image,
        data: NutritionData,
    ) -> None:
        min_title = self._pt(18)
        max_title = self._pt(36)

        title_max_w = max(
            10,
            self._content_width - self._pt(1),
        )

        self._title_px = fit_font_size(
            FDA_TITLE,
            title_max_w,
            bold=True,
            min_size=min_title,
            max_size=max_title,
        )

        draw_text(
            img,
            (self._left, self._cursor_y),
            FDA_TITLE,
            self._title_px,
            bold=True,
        )

        _, title_h = measure_text(
            FDA_TITLE,
            self._title_px,
            bold=True,
        )

        self._cursor_y += title_h + self._pt(1)

        servings = format_servings(
            data.serving_info.servings_per_container
        )

        servings_px = self._pt(FDA_SERVINGS_PT)

        draw_text(
            img,
            (self._left, self._cursor_y),
            servings,
            servings_px,
            bold=False,
        )

        _, servings_h = measure_text(
            servings,
            servings_px,
            bold=False,
        )

        self._cursor_y += (
            servings_h
            + self._pt(FDA_SERVINGS_LEADING_PT)
        )

        size_px = self._pt(FDA_SERVING_SIZE_PT)

        label = "Serving size"

        draw_text(
            img,
            (self._left, self._cursor_y),
            label,
            size_px,
            bold=True,
        )

        draw_text_right(
            img,
            self._right,
            self._cursor_y,
            data.serving_info.serving_size,
            size_px,
            bold=False,
        )

        _, size_h = measure_text(
            label,
            size_px,
            bold=True,
        )

        self._cursor_y += size_h + self._pt(2)

        self._rule(
            img,
            FDA_RULE_HEAVY_PT,
        )

    def _get_text_height(
        self,
        text: str,
        font_size: int,
        bold: bool = False,
    ) -> int:
        font = load_font(
            font_size,
            bold=bold,
        )

        dummy = Image.new(
            "RGB",
            (1, 1),
            RGB_WHITE,
        )

        draw = ImageDraw.Draw(dummy)

        bbox = draw.textbbox(
            (0, 0),
            text,
            font=font,
            anchor="lt",
        )

        return max(
            1,
            bbox[3] - bbox[1],
        )

    def _draw_calories(
        self,
        img: Image.Image,
        calories: int,
    ) -> None:
        self._cursor_y += self._pt(3)

        amount_px = self._pt(FDA_AMOUNT_PT)

        draw_text(
            img,
            (self._left, self._cursor_y),
            "Amount per serving",
            amount_px,
            bold=True,
        )

        amount_h = self._get_text_height(
            "Amount per serving",
            amount_px,
            bold=True,
        )

        self._cursor_y += amount_h + self._pt(1)

        cal_label_px = self._pt(
            FDA_CALORIES_LABEL_PT
        )

        cal_value_px = max(
            self._pt(FDA_CALORIES_VALUE_PT),
            int(self._title_px * 1.08),
        )

        cal_text = str(calories)

        label_h = self._get_text_height(
            "Calories",
            cal_label_px,
            bold=True,
        )

        value_h = self._get_text_height(
            cal_text,
            cal_value_px,
            bold=True,
        )

        row_h = max(
            label_h,
            value_h,
        )

        label_y = (
            self._cursor_y
            + row_h
            - label_h
        )

        value_y = (
            self._cursor_y
            + row_h
            - value_h
        )

        draw_text(
            img,
            (self._left, label_y),
            "Calories",
            cal_label_px,
            bold=True,
        )

        draw_text_right(
            img,
            self._right,
            value_y,
            cal_text,
            cal_value_px,
            bold=True,
        )

        # Tăng khoảng cách đệm (padding) từ Calories xuống đường kẻ dưới
        # để nhãn trông thanh thoát và bớt tù túng hơn.
        self._cursor_y += row_h + self._pt(4)

        self._rule(
            img,
            FDA_RULE_MEDIUM_PT,
        )

    def _hairline_gap(
        self,
        img: Image.Image,
        x: Optional[int] = None,
        width: Optional[int] = None,
    ) -> None:
        pad = self._pt(FDA_HAIRLINE_PAD_PT)

        self._cursor_y += pad

        self._rule(
            img,
            FDA_RULE_HAIRLINE_PT,
            x=x,
            width=width,
        )

        self._cursor_y += pad

    def _draw_nutrient_table(
        self,
        img: Image.Image,
        nutrients: List[Nutrient],
    ) -> None:
        # Tăng khoảng cách từ đường kẻ trung bình phía trên xuống chữ "% Daily Value*"
        self._cursor_y += self._pt(4)

        dv_px = self._pt(FDA_DV_HEADER_PT)
        header = "% Daily Value*"

        draw_text_right(
            img,
            self._right,
            self._cursor_y,
            header,
            dv_px,
            bold=True,
        )

        _, dv_h = measure_text(
            header,
            dv_px,
            bold=True,

"""Pydantic models for nutrition facts label data."""

from typing import List, Optional

from pydantic import BaseModel, Field, field_validator


class Nutrient(BaseModel):
    """A single nutrient row on the label."""

    name: str = Field(..., min_length=1, description='Nutrient name, e.g. "Total Fat"')
    value: float = Field(..., ge=0, description="Amount of the nutrient")
    unit: str = Field(..., min_length=1, description='Unit, e.g. "g", "mg", "μg"')
    daily_value_percent: Optional[float] = Field(
        default=None,
        ge=0,
        description="Percent Daily Value, e.g. 23 for 23%",
    )

    @field_validator("name", "unit")
    @classmethod
    def strip_text(cls, value: str) -> str:
        stripped = value.strip()
        if not stripped:
            raise ValueError("must not be empty or whitespace")
        return stripped


class ServingInfo(BaseModel):
    """Serving size and servings per container."""

    serving_size: str = Field(..., min_length=1, description='e.g. "2 tbsp (32g)"')
    servings_per_container: float = Field(..., gt=0)

    @field_validator("serving_size")
    @classmethod
    def strip_serving_size(cls, value: str) -> str:
        stripped = value.strip()
        if not stripped:
            raise ValueError("serving_size must not be empty or whitespace")
        return stripped


class NutritionData(BaseModel):
    """Complete nutrition facts input for one product label."""

    product_name: str = Field(..., min_length=1)
    serving_info: ServingInfo
    calories: int = Field(..., ge=0)
    nutrients: List[Nutrient] = Field(default_factory=list)
    manufacturer: Optional[str] = None

    @field_validator("product_name")
    @classmethod
    def strip_product_name(cls, value: str) -> str:
        stripped = value.strip()
        if not stripped:
            raise ValueError("product_name must not be empty or whitespace")
        return stripped

    @field_validator("manufacturer")
    @classmethod
    def strip_manufacturer(cls, value: Optional[str]) -> Optional[str]:
        if value is None:
            return None
        stripped = value.strip()
        return stripped or None

"""Pillow drawing helpers for Nutrition Facts labels."""

from typing import List, Optional, Tuple

from PIL import Image, ImageDraw

from nutrition_facts.constants import COLOR_BLACK
from nutrition_facts.utils import load_font


def _to_draw(img: Image.Image) -> ImageDraw.ImageDraw:
    return ImageDraw.Draw(img)


def draw_text(
    img: Image.Image,
    position: Tuple[int, int],
    text: str,
    font_size: int,
    bold: bool = False,
    color: str = COLOR_BLACK,
    anchor: str = "lt",
) -> None:
    """Draw text. Default anchor is top-left (Pillow 'lt')."""
    draw = _to_draw(img)
    font = load_font(font_size, bold=bold)
    draw.text(position, text, fill=color, font=font, anchor=anchor)


def draw_text_right(
    img: Image.Image,
    right_x: int,
    y: int,
    text: str,
    font_size: int,
    bold: bool = False,
    color: str = COLOR_BLACK,
    anchor: str = "rt",
) -> None:
    """Draw text right-aligned to right_x. Default anchor is top-right."""
    draw = _to_draw(img)
    font = load_font(font_size, bold=bold)
    draw.text((right_x, y), text, fill=color, font=font, anchor=anchor)


def measure_text(text: str, font_size: int, bold: bool = False) -> Tuple[int, int]:
    """Return (advance width, line height) using font metrics."""
    font = load_font(font_size, bold=bold)
    if hasattr(font, "getlength"):
        width = int(round(font.getlength(text)))
    else:
        dummy = Image.new("RGB", (8, 8))
        draw = ImageDraw.Draw(dummy)
        bbox = draw.textbbox((0, 0), text, font=font)
        width = bbox[2] - bbox[0]
    ascent, descent = font.getmetrics()
    return width, ascent + descent


def fit_font_size(
    text: str,
    max_width: int,
    bold: bool,
    min_size: int,
    max_size: int,
) -> int:
    """Largest font size in [min_size, max_size] whose text fits max_width."""
    best = min_size
    low, high = min_size, max_size
    while low <= high:
        mid = (low + high) // 2
        width, _ = measure_text(text, mid, bold=bold)
        if width <= max_width:
            best = mid
            low = mid + 1
        else:
            high = mid - 1
    return best


def wrap_text(text: str, font_size: int, bold: bool, max_width: int) -> List[str]:
    """Word-wrap text to max_width pixels."""
    words = text.split()
    if not words:
        return []
    lines: List[str] = []
    current = words[0]
    for word in words[1:]:
        trial = f"{current} {word}"
        width, _ = measure_text(trial, font_size, bold=bold)
        if width <= max_width:
            current = trial
        else:
            lines.append(current)
            current = word
    lines.append(current)
    return lines


def draw_horizontal_line(
    img: Image.Image,
    y: int,
    width: int,
    thickness: int = 1,
    color: str = COLOR_BLACK,
    x: int = 0,
) -> None:
    """Draw a horizontal bar from x to x+width at y."""
    draw = _to_draw(img)
    thickness = max(1, thickness)
    draw.rectangle([x, y, x + width - 1, y + thickness - 1], fill=color)


def draw_border(img: Image.Image, thickness: int = 1, color: str = COLOR_BLACK) -> None:
    """Draw a rectangle border around the image."""
    draw = _to_draw(img)
    w, h = img.size
    for i in range(max(1, thickness)):
        draw.rectangle([i, i, w - 1 - i, h - 1 - i], outline=color)


def draw_nutrient_row(
    img: Image.Image,
    y: int,
    label: str,
    value: str,
    daily_value: Optional[str] = None,
    indent: int = 0,
    name_bold: bool = True,
    font_size: int = 12,
    left: Optional[int] = None,
    right: Optional[int] = None,
) -> int:
    """Draw one nutrient line: bold/regular name, regular amount, bold %DV.

    Returns the text height used.
    """
    if left is None:
        left = 16
    if right is None:
        right = img.width - 16
    x = left + indent
    draw_text(img, (x, y), label, font_size, bold=name_bold)
    name_w, name_h = measure_text(label, font_size, bold=name_bold)
    value_h = 0
    if value:
        gap, _ = measure_text(" ", font_size, bold=False)
        draw_text(img, (x + name_w + gap, y), value, font_size, bold=False)
        _, value_h = measure_text(value, font_size, bold=False)
    if daily_value:
        draw_text_right(img, right, y, daily_value, font_size, bold=True)
    return max(name_h, value_h, 1)

def format_nutrient_value(value: float, unit: str) -> str:
    """Format a nutrient amount, dropping trailing zeros for whole numbers."""
    if value == int(value):
        amount = str(int(value))
    else:
        amount = f"{value:g}"
    return f"{amount}{unit}"


def format_daily_value(percent: Optional[float]) -> Optional[str]:
    """Return a daily-value string like '23%', or None if not provided."""
    if percent is None:
        return None
    if percent == int(percent):
        return f"{int(percent)}%"
    return f"{percent:g}%"


def format_servings(count: float) -> str:
    """Format the FDA servings line, e.g. '14 servings per container'."""
    if count == int(count):
        number = str(int(count))
        n = int(count)
    else:
        number = f"{count:g}"
        n = count
    noun = "serving" if n == 1 else "servings"
    return f"{number} {noun} per container"


def normalize_nutrient_key(name: str) -> str:
    return " ".join(name.strip().lower().split())


def nutrient_role(name: str) -> str:
    """Return 'micro', 'added', 'sub', 'main', or 'other'."""
    key = normalize_nutrient_key(name)
    if key in MICRO_NUTRIENTS:
        return "micro"
    if key in ADDED_SUGARS_NAMES:
        return "added"
    if key in SUB_NUTRIENTS:
        return "sub"
    if key in MAIN_NUTRIENTS:
        return "main"
    return "other"


def display_nutrient_name(name: str) -> str:
    key = normalize_nutrient_key(name)
    if key in DISPLAY_NAMES:
        return DISPLAY_NAMES[key]
    if key in MICRO_NUTRIENTS:
        return MICRO_NUTRIENTS[key]
    return name


def sort_nutrients(nutrients: List[Nutrient]) -> List[Nutrient]:
    """Sort nutrients by FDA-style order; unknown names keep relative order at the end."""
    order = {name.lower(): index for index, name in enumerate(STANDARD_NUTRIENT_ORDER)}
    known = [n for n in nutrients if n.name.lower() in order]
    unknown = [n for n in nutrients if n.name.lower() not in order]
    known.sort(key=lambda item: order[item.name.lower()])
    return known + unknown


def partition_nutrients(nutrients: List[Nutrient]) -> Tuple[List[Nutrient], List[Nutrient]]:
    """Split into (macronutrient rows, vitamin/mineral rows) in FDA order."""
    ordered = sort_nutrients(nutrients)
    macros = [n for n in ordered if nutrient_role(n.name) != "micro"]
    micros = [n for n in ordered if nutrient_role(n.name) == "micro"]
    return macros, micros


def needs_full_hairline_before(prev_name: str, curr_name: str) -> bool:
    """Return True if a full-width hairline rule should precede curr_name.

    FDA rules (priority order):
      1. Before Protein (final main)  -> never a hairline
      2. Within FAT_SUB_GROUP (Total Fat / Saturated / Trans)  -> no hairline
      3. Carb parent -> first carb sub (Dietary Fiber / Total Sugars / Sugars)  -> no hairline
      4. Otherwise -> hairline needed.
    """
    prev = normalize_nutrient_key(prev_name)
    curr = normalize_nutrient_key(curr_name)

    # 1. Protein is the last main macro — FDA draws no hairline above it.
    if curr == "protein":
        return False

    # 2. Fat sub-group members stay grouped without hairlines.
    if prev in FAT_SUB_GROUP and curr in FAT_SUB_GROUP:
        return False

    # 3. Transitioning from a Carb parent directly to one of its sub-nutrients.
    if prev in CARB_PARENTS and curr in CARB_SUB_GROUP:
        return False

    return True

def test_needs_full_hairline_before_fda_rules():
    from nutrition_facts.utils import needs_full_hairline_before

    # Fat group: no hairlines between Total Fat / Saturated Fat / Trans Fat
    assert needs_full_hairline_before("Total Fat", "Saturated Fat") is False
    assert needs_full_hairline_before("Saturated Fat", "Trans Fat") is False
    assert needs_full_hairline_before("Total Fat", "Trans Fat") is False

    # Trans Fat -> next main nutrient (Cholesterol) needs hairline
    assert needs_full_hairline_before("Trans Fat", "Cholesterol") is True

    # Main -> main needs hairlines
    assert needs_full_hairline_before("Cholesterol", "Sodium") is True
    assert needs_full_hairline_before("Sodium", "Total Carbohydrate") is True

    # Carb parent -> first sub (Dietary Fiber): no hairline
    assert needs_full_hairline_before("Total Carbohydrate", "Dietary Fiber") is False
    assert needs_full_hairline_before("Carbohydrates", "Dietary Fiber") is False

    # Dietary Fiber -> Total Sugars (sibling carb subs): needs hairline
    assert needs_full_hairline_before("Dietary Fiber", "Total Sugars") is True

    # Before Protein (final main): NO hairline, regardless of predecessor
    assert needs_full_hairline_before("Added Sugars", "Protein") is False
    assert needs_full_hairline_before("Total Sugars", "Protein") is False
    assert needs_full_hairline_before("Sodium", "Protein") is False

    # Typical valid transition (Added Sugars is role="added" handled separately)
    assert needs_full_hairline_before("Total Sugars", "Added Sugars") is True
